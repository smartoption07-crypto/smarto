
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans, AgglomerativeClustering
from scipy.cluster.hierarchy import dendrogram, linkage

df = pd.read_csv("Mall_Customers.csv")

print("Dataset Shape:", df.shape)
print("\nFirst Five Records:")
print(df.head())

df.drop('CustomerID', axis=1, inplace=True)

df['Gender'] = df['Gender'].map({'Male': 1, 'Female': 0})

print("\nAfter Encoding:")
print(df.head())

scaler = StandardScaler()
X = scaler.fit_transform(df)

print("\nScaled Dataset Shape:", X.shape)

wcss = []

for i in range(1, 11):
    km = KMeans(n_clusters=i, random_state=42, n_init=10)
    km.fit(X)
    wcss.append(km.inertia_)

plt.figure(figsize=(7, 4))
plt.plot(range(1, 11), wcss, marker='o')
plt.title("Elbow Method")
plt.xlabel("Number of Clusters")
plt.ylabel("WCSS")
plt.grid()
plt.show()

kmeans = KMeans(n_clusters=5, random_state=42, n_init=10)
df['KMeans_Cluster'] = kmeans.fit_predict(X)

print("\nK-Means Clustering")
print(df['KMeans_Cluster'].value_counts().sort_index())

hierarchical = AgglomerativeClustering(n_clusters=5)
df['Hierarchical_Cluster'] = hierarchical.fit_predict(X)

print("\nHierarchical Clustering")
print(df['Hierarchical_Cluster'].value_counts().sort_index())

Z = linkage(X, method='ward')

plt.figure(figsize=(10, 5))
dendrogram(Z, truncate_mode='lastp', p=30)
plt.title("Hierarchical Clustering Dendrogram")
plt.xlabel("Data Points / Clusters")
plt.ylabel("Distance")
plt.show()

plt.figure(figsize=(7, 5))
sns.scatterplot(
    x=df['Annual Income (k$)'],
    y=df['Spending Score (1-100)'],
    hue=df['KMeans_Cluster'],
    palette='viridis',
    s=70
)
plt.title("K-Means Clustering")
plt.show()

plt.figure(figsize=(7, 5))
sns.scatterplot(
    x=df['Annual Income (k$)'],
    y=df['Spending Score (1-100)'],
    hue=df['Hierarchical_Cluster'],
    palette='rainbow',
    s=70
)
plt.title("Hierarchical Clustering")
plt.show()

print("\nK-Means Cluster Analysis:")
print(df.groupby('KMeans_Cluster')[
    ['Age', 'Annual Income (k$)', 'Spending Score (1-100)']
].mean().round(2))

print("\nHierarchical Cluster Analysis:")
print(df.groupby('Hierarchical_Cluster')[
    ['Age', 'Annual Income (k$)', 'Spending Score (1-100)']
].mean().round(2))

print("\nClustering Completed Successfully!")
