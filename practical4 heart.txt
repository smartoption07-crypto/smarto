
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import statistics as s

from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler

sns.set_style("whitegrid")

df = pd.read_csv("heart.csv")

print("First Five Records:")
print(df.head())

print("\nDataset Information:")
df.info()

print("\nColumn Names:")
print(df.columns)

print("\nDataset Shape:")
print(df.shape)

print("\nDuplicate Records:", df.duplicated().sum())
df.drop_duplicates(inplace=True)
print("\nMissing Values Before Handling:")
print(df.isnull().sum())

for col in df.columns:
    df[col] = df[col].fillna(df[col].median())

print("\nMissing Values After Handling:")
print(df.isnull().sum())

print("\nEncoded Sex Values:")
print(df['sex'].unique())

print("\nStatistical Summary:")
print(df.describe())

print("\nMean:")
print(df.mean(numeric_only=True))

print("\nMedian:")
print(df.median(numeric_only=True))

print("\nStandard Deviation:")
print(df.std(numeric_only=True))

print("\nCorrelation Matrix:")
print(df.corr())
plt.figure(figsize=(8, 5))
plt.hist(df['age'], bins=10, edgecolor='black')
plt.title("Histogram of Age")
plt.xlabel("Age")
plt.ylabel("Frequency")
plt.show()

plt.figure(figsize=(8, 5))
plt.hist(df['chol'], bins=10, edgecolor='black')
plt.title("Histogram of Cholesterol")
plt.xlabel("Cholesterol")
plt.ylabel("Frequency")
plt.show()

plt.figure(figsize=(8, 5))
sns.boxplot(x=df['age'])
plt.title("Boxplot of Age")
plt.show()

plt.figure(figsize=(8, 5))
sns.boxplot(x=df['chol'])
plt.title("Boxplot of Cholesterol")
plt.show()

print("\nAverage Age:", s.mean(df['age']))

# 8. Feature Scaling and Train-Test Split
X = df.drop('target', axis=1)
y = df['target']

X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.2, random_state=56
)

scaler = StandardScaler()

X_train = scaler.fit_transform(X_train)
X_test = scaler.transform(X_test)

print("\nAfter Feature Scaling:")
print(X_train)

print("\nTraining Data Shape:", X_train.shape)
print("Testing Data Shape:", X_test.shape)

print("\nTraining Target Shape:", y_train.shape)
print("Testing Target Shape:", y_test.shape)

print("\nData Preprocessing Completed Successfully!")
